{{- /* Member photo resolved at build time; falls back to an initials avatar. Params: page, size (px), class Looked up as media/images/members/.jpg, then .jpg (for members without GitHub) */ -}} {{- $page := .page -}} {{- $path := "" -}} {{- range slice $page.Params.github $page.File.ContentBaseName -}} {{- $candidate := printf "media/images/members/%s.jpg" . -}} {{- if and . (not $path) (fileExists (printf "static/%s" $candidate)) -}} {{- $path = $candidate -}} {{- end -}} {{- end -}} {{- if $path -}} {{- else -}} {{- $words := split (trim $page.Title " ") " " -}} {{- $initials := substr (index $words 0) 0 1 -}} {{- if gt (len $words) 1 -}} {{- $initials = printf "%s%s" $initials (substr (index $words (sub (len $words) 1)) 0 1) -}} {{- end -}} {{ upper $initials }} {{- end -}}
